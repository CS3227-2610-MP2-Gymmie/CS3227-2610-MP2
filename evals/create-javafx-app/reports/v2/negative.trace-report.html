package gymmie;

import java.io.IOException;

import javafx.application.Application;
import javafx.fxml.FXMLLoader;
import javafx.scene.Scene;
import javafx.stage.Stage;

/**
 * Starts the Gymmie JavaFX application.
 */
public class App extends Application {

    /**
     * Builds and displays the primary application window.
     *
     * @param stage the primary stage supplied by JavaFX
     * @throws IOException if the FXML view cannot be loaded
     */
    @Override
    public void start(Stage stage) throws IOException {
        FXMLLoader loader = new FXMLLoader(App.class.getResource("/gymmie/view/MainWindow.fxml"));
        Scene scene = new Scene(loader.load(), 640, 420);
        scene.getStylesheets().add(App.class.getResource("/gymmie/css/gymmie.css").toExternalForm());

        stage.setTitle("Gymmie");
        stage.setScene(scene);
        stage.setMinWidth(480);
        stage.setMinHeight(320);
        stage.show();
    }
}
